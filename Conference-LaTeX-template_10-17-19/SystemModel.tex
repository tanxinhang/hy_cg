\section{System Model}

\label{sec:system_model}

We consider a sensing-centric multi-UAV OTFS-ISAC network, as shown in
Fig.~\ref{fig:system_framework}(a). The UAV and target sets are denoted by
$\mathcal M=\{1,\ldots,M\}$ and $\mathcal Q=\{1,\ldots,Q\}$, respectively.
UAV $i$ and target $q$ have position--velocity pairs
$(\mathbf p_i,\mathbf v_i)$ and $(\mathbf p_q,\mathbf v_q)$. A cooperative
candidate is indexed by the target-dependent triplet $(i,j,q)$, where UAV $i$
transmits an OTFS-ISAC probing signal, target $q$ reflects it, and UAV $j$
receives the bistatic echo. The inter-UAV reporting topology is represented by
a directed graph $\mathcal G=(\mathcal M,\mathcal E)$, where
$(i,j)\in\mathcal E$ if $\|\mathbf p_i-\mathbf p_j\|\le d_{\max}$ and
$i\ne j$. In the scheduling model, the same ordered pair $(i,j)$ indexes the bistatic
sensing pair and the effective reporting relation associated with this
candidate. We use a geometry-driven soft-information abstraction rather than a full
waveform-level OTFS receiver model. The OTFS structure enters the scheduling
problem through DD-bin validity, fractional DD energy loss, same-bin collision,
and local DD-grid refinement. These DD-domain quantities determine whether a
target-dependent triplet provides useful soft information after reporting and
fusion.

\subsection{Communication and Reporting Model}
\label{subsec:comm_model}

The transmit power of UAV $i$ is split into sensing and communication parts,
i.e.,
\begin{equation}
	P_i^{\mathrm s}=\rho P_i,\qquad
	P_i^{\mathrm c}=(1-\rho)P_i .
	\label{eq:power_split}
\end{equation}
For a directed inter-UAV link $(a,b)$, the communication-channel power gain is
\begin{equation}
	g_{ab}^{\mathrm c}
	=
	\left(\frac{\lambda}{4\pi}\right)^2
	d_{ab}^{-\alpha_{\mathrm c}}
	\zeta_{ab}|h_{ab}^{\mathrm c}|^2,
	\qquad
	d_{ab}=\|\mathbf p_a-\mathbf p_b\|.
	\label{eq:comm_channel_gain}
\end{equation}
Here $\lambda$, $\alpha_{\mathrm c}$, $\zeta_{ab}$, and $h_{ab}^{\mathrm c}$
denote the carrier wavelength, path-loss exponent, shadowing, and small-scale
fading, respectively.

The communication SINR from UAV $i$ to UAV $j$ is modeled as
\begin{equation}
	\gamma_{ij}^{\mathrm c}
	=
	\frac{P_i^{\mathrm c}g_{ij}^{\mathrm c}}
	{N_0+
		\sum_{k\in\mathcal M\setminus\{i,j\}}
		\left(P_k^{\mathrm c}+\epsilon_{\mathrm s}P_k^{\mathrm s}\right)
		g_{kj}^{\mathrm c}
		+\epsilon_{\mathrm d}P_i^{\mathrm s}g_{ij}^{\mathrm c}} .
	\label{eq:comm_sinr}
\end{equation}
The coefficients $\epsilon_{\mathrm s}$ and $\epsilon_{\mathrm d}$ represent
neighbor-induced sensing-to-communication leakage and intra-link leakage from
the sensing component of UAV $i$ to the communication receiver, respectively.

With communication bandwidth $B_{\mathrm c}=N\Delta f$, the reporting rate,
link reliability, and exchange delay are
\begin{equation}
	R_{ij}=B_{\mathrm c}\log_2(1+\gamma_{ij}^{\mathrm c}),\qquad
	\chi_{ij}
	=
	\frac{\gamma_{ij}^{\mathrm c}}
	{\gamma_{ij}^{\mathrm c}+\gamma_{\mathrm{req}}},
	\qquad
	c_{ijq}=\frac{B_q}{R_{ij}},
	\label{eq:rate_reliability_delay}
\end{equation}
where $\gamma_{\mathrm{req}}=2^{R_{\min}/B_{\mathrm c}}-1$ and
$B_q=K_{\mathrm{pkt}}b_{\mathrm d}$ is the soft-information payload size. A
reporting link is feasible when $R_{ij}\ge R_{\min}$.

\subsection{DD-Domain Bistatic Sensing Model}
\label{subsec:dd_sensing_model}

For triplet $(i,j,q)$, define
$d_{iq}=\|\mathbf p_q-\mathbf p_i\|$,
$d_{jq}=\|\mathbf p_q-\mathbf p_j\|$,
$\mathbf u_{iq}=(\mathbf p_q-\mathbf p_i)/d_{iq}$, and
$\mathbf u_{jq}=(\mathbf p_q-\mathbf p_j)/d_{jq}$. The bistatic delay and
Doppler shift are
\begin{equation}
	\tau_{ijq}=\frac{d_{iq}+d_{jq}}{c},\qquad
	\nu_{ijq}
	=
	\frac{
		(\mathbf v_q-\mathbf v_i)^T\mathbf u_{iq}
		+
		(\mathbf v_q-\mathbf v_j)^T\mathbf u_{jq}}
	{\lambda}.
	\label{eq:bistatic_delay_doppler}
\end{equation}

The corresponding DD-grid indices are
\begin{equation}
	l_{ijq}=\operatorname{round}(\tau_{ijq}L\Delta f),\qquad
	k_{ijq}=\operatorname{round}(\nu_{ijq}NT).
	\label{eq:dd_grid_mapping}
\end{equation}
A triplet is DD-valid if $0\le l_{ijq}<L$ and
$-N/2\le k_{ijq}<N/2$. The fractional DD offsets are
\begin{equation}
	\varepsilon_{ijq}^{\tau}=\tau_{ijq}L\Delta f-l_{ijq},\qquad
	\varepsilon_{ijq}^{\nu}=\nu_{ijq}NT-k_{ijq}.
	\label{eq:dd_fractional_offsets}
\end{equation}

Using the normalized sinc function, the coarse DD-domain gain is
\begin{equation}
	\eta_{ijq}^{\mathrm c}
	=
	\operatorname{sinc}^2(\varepsilon_{ijq}^{\tau})
	\operatorname{sinc}^2(\varepsilon_{ijq}^{\nu}).
	\label{eq:coarse_dd_gain}
\end{equation}
For C2F DD refinement, a local $(2W+1)\times(2W+1)$ DD neighborhood provides
the recoverable energy $\eta_{ijq}^{\mathrm{loc}}$, and the refined gain is
\begin{equation}
	\eta_{ijq}^{\mathrm f}
	=
	\min\!\left\{
	1,\,
	\max\!\left[
	\eta_{\min},
	\eta_{ijq}^{\mathrm c}
	+
	\kappa_{\mathrm{dd}}
	(\eta_{ijq}^{\mathrm{loc}}-\eta_{ijq}^{\mathrm c})
	\right]
	\right\}.
	\label{eq:fine_dd_gain}
\end{equation}
Here $\kappa_{\mathrm{dd}}\in[0,1]$ controls the refinement strength and
$\eta_{\min}$ is the retained-energy floor.

If multiple DD-valid targets observed by the same UAV pair fall into the same
DD bin, we use the collision penalty
\begin{equation}
	\eta_{ijq}^{\mathrm{col}}
	=
	\frac{1}{n_{ijq}^{\alpha_{\mathrm{col}}}},
	\label{eq:dd_collision_penalty}
\end{equation}
where $n_{ijq}$ is the number of targets sharing bin $(l_{ijq},k_{ijq})$.

The bistatic target gain is
\begin{equation}
	g_{ijq}^{\mathrm s}
	=
	\frac{\lambda^2\sigma_q}{(4\pi)^3d_{iq}^2d_{jq}^2},
	\label{eq:bistatic_target_gain}
\end{equation}
where $\sigma_q$ denotes the target RCS. Combining bistatic gain, processing
gain, DD-domain gain, and collision penalty gives
\begin{equation}
	\gamma_{ijq}^{\mathrm s}
	=
	\frac{
		\widetilde P_i^{\mathrm s}
		g_{ijq}^{\mathrm s}
		G_{\mathrm p}
		\eta_{ijq}^{\mathrm{DD}}
		\eta_{ijq}^{\mathrm{col}}
	}
	{N_0+I_{ij}^{\mathrm{res}}},
	\qquad
	\eta_{ijq}^{\mathrm{DD}}
	\in
	\{\eta_{ijq}^{\mathrm c},\eta_{ijq}^{\mathrm f}\}.
	\label{eq:sensing_sinr}
\end{equation}
In the default sensing-only mode, $\widetilde P_i^{\mathrm s}=P_i^{\mathrm s}$,
and $G_{\mathrm p}=NL$. The residual term $I_{ij}^{\mathrm{res}}$ captures
self-interference, direct leakage, and multi-UAV sensing leakage at the sensing
receiver.

\subsection{Soft-Information Fusion Model}
\label{subsec:soft_fusion_model}

Each feasible triplet produces a scalar soft statistic $s_{ijq}$. Its
target-present mean and $\mathcal H_0$-side variance are modeled as
\begin{equation}
	\mu_{ijq}
	=
	\kappa_\mu\log(1+\gamma_{ijq}^{\mathrm s}),
	\qquad
	\sigma_{ij,0}^2
	=
	\sigma_0^2
	\left(1+\xi I_{ij}^{\mathrm{res}}/N_0\right).
	\label{eq:soft_stat_model}
\end{equation}
Communication-error calibration modifies the statistics used for fusion as
\begin{equation}
	\mu_{ijq}^{\mathrm{eff}}=\chi_{ij}\mu_{ijq},\qquad
	\bar{\sigma}_{ij,0}^2
	=
	\chi_{ij}\sigma_{ij,0}^2
	+
	(1-\chi_{ij})\sigma_{\mathrm{err}}^2.
	\label{eq:comm_error_calibration}
\end{equation}

For selected set $\mathcal S_q$, the fused statistic is
\begin{equation}
	F_q
	=
	\sum_{(i,j)\in\mathcal S_q}
	w_{ijq}s_{ijq},
	\qquad
	w_{ijq}
	\propto
	\frac{\mu_{ijq}^{\mathrm{eff}}}{\bar{\sigma}_{ij,0}^2}.
	\label{eq:fused_statistic}
\end{equation}
The corresponding composite deflection is
\begin{equation}
	D_q(\mathcal S_q)
	=
	\frac{
		\left(
		\sum_{(i,j)\in\mathcal S_q}
		w_{ijq}\mu_{ijq}^{\mathrm{eff}}
		\right)^2}
	{
		\sum_{(i,j)\in\mathcal S_q}
		w_{ijq}^2\bar{\sigma}_{ij,0}^2
	}.
	\label{eq:composite_deflection}
\end{equation}

Approximating $F_q$ as Gaussian, the detection probability is
\begin{equation}
	P_{D,q}
	\approx
	\mathsf Q
	\left(
	\mathsf Q^{-1}(P_{\mathrm{FA}})
	-
	\sqrt{D_q(\mathcal S_q)}
	\right),
	\label{eq:pd_approx}
\end{equation}
where $\mathsf Q(\cdot)$ denotes the Gaussian $Q$-function.
